cmake_minimum_required(VERSION 3.20)
project(KineticTracker VERSION 2.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs video videoio dnn highgui)
add_library(kinetic_core src/tracker.cpp src/detector.cpp)
target_include_directories(kinetic_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(kinetic_core PUBLIC ${OpenCV_LIBS})
add_executable(kinetic_tracker src/main.cpp)
target_link_libraries(kinetic_tracker PRIVATE kinetic_core)
if(MSVC)
  target_compile_options(kinetic_core PRIVATE /W4)
  target_compile_options(kinetic_tracker PRIVATE /W4)
else()
  target_compile_options(kinetic_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(kinetic_tracker PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_tracker tests/test_tracker.cpp)
  target_link_libraries(test_tracker PRIVATE kinetic_core)
  add_test(NAME tracker_semantics COMMAND test_tracker)
  add_executable(test_scheduler tests/test_scheduler.cpp)
  target_include_directories(test_scheduler PRIVATE include)
  add_test(NAME scheduler_semantics COMMAND test_scheduler)
  add_executable(test_detector tests/test_detector.cpp)
  target_link_libraries(test_detector PRIVATE kinetic_core)
  add_test(NAME detector_semantics COMMAND test_detector)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_Interpreter_FOUND)
    add_test(NAME python_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -p test_*.py)
    set_tests_properties(python_tests PROPERTIES
      WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
      ENVIRONMENT "KINETIC_TRACKER_EXE=$<TARGET_FILE:kinetic_tracker>")
  endif()
endif()
